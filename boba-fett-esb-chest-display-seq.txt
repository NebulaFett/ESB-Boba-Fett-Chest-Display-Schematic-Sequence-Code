BOBA FETT ESB CHEST DISPLAY
ATmega328 Code

Description:
This code controls the LED sequence for a DIY Boba Fett ESB chest display using an ATmega328 microcontroller.

The display sequence is based on research by RafalFett:
https://www.thedentedhelmet.com/forums/threads/esb-chest-display-sequence.20061/

The code is my own implementation based on the research referenced above.

This project is shared for hobbyist and educational purposes. The code is provided "AS IS", without any warranty. Use it at your own risk.

If you find an easier or more efficient way to implement the sequence, feedback and suggestions are welcome.

const byte sequenceLength =
sizeof(sequence) / sizeof(sequence[0]);

byte currentStep = 0;

unsigned long stepStart = 0;

byte currentDisplay = 0;
unsigned long muxTimer = 0;

void allDisplaysOff()
{
  for(byte i=0;i<5;i++)
    digitalWrite(displayPins[i],LOW);
}

void setSegments(byte digit)
{
  for(byte s=0;s<7;s++)
  {
    digitalWrite(segmentPins[s],
                  sequence[currentStep].digits[digit][s]);
  }
}

void refreshDisplay()
{

  if(millis()-muxTimer < 2)
    return;

  muxTimer=millis();

  allDisplaysOff();

  setSegments(currentDisplay);

  digitalWrite(displayPins[currentDisplay],HIGH);

  currentDisplay++;

  if(currentDisplay>=5)
    currentDisplay=0;
}



void setLEDs()
{
  for(byte i=0;i<5;i++)
    digitalWrite(ledPins[i],LOW);

  for(byte i=0;i<5;i++)
  {
    if(sequence[currentStep].leds[i])
    {
      digitalWrite(ledPins[i],HIGH);
      break;
    }
  }
}

void nextStep()
{
  currentStep++;

  if(currentStep>=sequenceLength)
    currentStep=0;

  setLEDs();

  stepStart=millis();
}

void setup()
{
  for(byte i=0;i<7;i++)
    pinMode(segmentPins[i],OUTPUT);


  for(byte i=0;i<5;i++)
  {
    pinMode(displayPins[i],OUTPUT);
    pinMode(ledPins[i],OUTPUT);
  }

  allDisplaysOff();
  setLEDs();

  stepStart=millis();
}

void loop()
{
  refreshDisplay();

  if(millis()-stepStart >= sequence[currentStep].timeMs)
  {
    nextStep();
  }
}
